#### 3.1 Provider — `lib/features/tracking/data/tracked_trip_provider.dart`

```dart
/// PRIVACY: toString never prints the destination name or any distance.
@immutable
final class TrackedTrip {
  const new({
    required this.config,            // TripConfigMsg (không toạ độ)
    required this.simulation,        // SimulationMsg?
    required this.state,             // TripStateMsg mới nhất
    required this.receivedAtMs,      // giờ thật (clock) lúc Dart nhận [state]
    this.acquiringSinceMs,           // giờ thật lần đầu thấy phase ACQUIRING của chuyến này trong phiên; null khi pha khác
    this.neverAcquired = false,      // đã thấy GpsLostMsg(neverAcquired) trong phiên, xoá khi GpsRecoveredMsg / gps != noFixYet
    this.lowBattery,                 // LowBatteryRiskMsg gần nhất trong phiên (availableMin, neededMin)
  });
  String get tripId => config.tripId;
  bool get simulated => simulation != null;
  TrackedTrip copyWith({...});
  @override String toString() => 'TrackedTrip(${state.phase.name})';
}

sealed class TrackingStatus { const new(); }
final class TrackingActive extends TrackingStatus { const new(this.trip); final TrackedTrip trip; }
final class TrackingEnded extends TrackingStatus {   // chuyến vừa kết thúc (đã thấy nó đang chạy hoặc snapshot ENDED)
  const new({required this.tripId, this.outcome, this.endReason}); final String tripId;
  final TripOutcomeMsg? outcome; final EndReasonMsg? endReason; }
final class TrackingNone extends TrackingStatus { const new(); }       // không có chuyến
final class TrackingError extends TrackingStatus { const new(); }      // currentTrip() lỗi (BridgeException / khác)

final trackedTripProvider = StreamProvider.autoDispose<TrackingStatus>((ref) =>
    trackedTripStream(ref.watch(coreBridgeProvider), nowMs: () => ref.read(clockProvider)().millisecondsSinceEpoch));

/// Exposed for unit tests (P01–P12).
Stream<TrackingStatus> trackedTripStream(CoreBridge bridge, {required int Function() nowMs});
```

Thuật toán (cùng khung `activeTripStream`: `StreamController` với `onListen: follow`, `onPause: unfollow`, `onResume: follow`,
`onCancel: unfollow + close`; mỗi bước chạy tuần tự trong `steps`, lỗi một bước → `reportNonFatal('trackedTripProvider', …)` chỉ kiểu lỗi):

```
follow(): updates ??= bridge.updates.listen(u => enqueue(apply(u)), onError: ignore)
          enqueue(loadSnapshot)
loadSnapshot():
   try snap = await bridge.currentTrip()
   catch BridgeException | Object → emit(TrackingError) (Object khác: reportNonFatal); return
   emit(fromSnapshot(snap))
fromSnapshot(snap):
   snap == null                       → TrackingNone
   snap.state.phase == ended          → TrackingEnded(tripId, outcome: snap.state.outcome)
   else                               → TrackingActive(TrackedTrip(config, simulation, state, receivedAtMs: now,
                                          acquiringSinceMs: phase == acquiring ? now : null))
apply(u), cur = last is TrackingActive ? last.trip : null:
   TripStateUpdateMsg(state):
     state.phase == ended                         → emit(TrackingEnded(state.tripId, outcome: state.outcome))
     cur == null || state.tripId != cur.tripId    → await loadSnapshot()      // cần config của chuyến mới
     state.atMs < cur.state.atMs                  → bỏ qua (bản phát lại cũ hơn snapshot; đồng hồ chuyến không lùi, T10 §2.6)
     else → emit(Active(cur.copyWith(state, receivedAtMs: now,
                 acquiringSinceMs: state.phase == acquiring ? (cur.acquiringSinceMs ?? now) : null,
                 neverAcquired: state.gps == noFixYet && cur.neverAcquired)))
   GpsLostMsg(tripId == cur?.tripId, cause == neverAcquired) → emit(Active(cur.copyWith(neverAcquired: true)))
   GpsRecoveredMsg(tripId == cur?.tripId)                    → emit(Active(cur.copyWith(neverAcquired: false)))
   LowBatteryRiskMsg(tripId == cur?.tripId)                  → emit(Active(cur.copyWith(lowBattery: u)))
   TripEndedMsg(trip) when trip.tripId == cur?.tripId        → emit(TrackingEnded(trip.tripId, outcome: trip.outcome,
                                                                 endReason: trip.endReason))
   khác → bỏ qua
```
- `TrackingEnded` đã phát thì `TripStateUpdateMsg` của cùng `tripId` sau đó bỏ qua (không có, T15 §5.11, nhưng phòng hờ).
- Riverpod 3 tạm dừng provider khi không còn listener hiển thị [RP-PAUSE]; pause huỷ subscription nên không đệm hàng giờ update.

#### 3.2 Sửa `CoreBridge.updates` (T14 F2) — `lib/core/bridge/core_bridge.dart`

```
late final Stream<TripUpdateMsg> _source = _broadcast(_updatesFactory());
StreamSubscription<TripUpdateMsg>? _memory;
updates = Stream.multi((controller) {
  if (_subscribers++ == 0) { _memory = _source.listen(_remember, onError: (_) {}); }   // listener ĐẦU TIÊN của source
  final latest = _latestState; if (latest != null) controller.add(latest);
  final sub = _source.listen(controller.add, onError: controller.addError, onDone: controller.closeSync);
  controller..onPause = sub.pause ..onResume = sub.resume      // pause KHÔNG chạm _memory
            ..onCancel = () { if (--_subscribers == 0) { _memory?.cancel(); _memory = null; _latestState = null; }
                              return sub.cancel(); };
}, isBroadcast: true);
```
- `_remember` không còn chạy trong `onData` của từng subscriber. `_memory` đăng ký trước mọi `sub` nên mỗi sự kiện được ghi nhớ trước khi tới
  subscriber (broadcast stream gửi theo thứ tự đăng ký).
- Hành vi khác giữ nguyên (không phát lại sự kiện, bỏ sau `TripEndedMsg` cùng `tripId`, xoá khi không còn ai nghe). Test "R2-2 (known
  limitation…)" trong `test/core/bridge/core_bridge_tester_test.dart` **đổi kỳ vọng**: subscriber mới nhận view mới nhất (đặt tên lại "R2-2 fixed (T26)").
